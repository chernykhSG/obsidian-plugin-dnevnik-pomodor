**Goal:** Создать Obsidian-плагин `dnevnik-pomodoro`, который закрывает все функциональные требования ТЗ «Дневник работы» (таймер 30/5 в статус-баре и плавающем окне, авто-открытие окна сводки по завершении, запись блока в ежедневную заметку, статистика в собственном окне).

```ts
// src/core.ts — чистая логика без импорта Obsidian. Покрыта unit-тестами.

export interface PomodoroBlock {
  start: string; // "HH:mm"
  end: string;   // "HH:mm"
  heading: string; // "## 🍅 HH:mm — HH:mm"
}

export function pad2(n: number): string {
  return n < 10 ? "0" + n : "" + n;
}

export function formatHHmm(d: Date): string {
  return pad2(d.getHours()) + ":" + pad2(d.getMinutes());
}

// Превращает "HH:mm" в Date на том же календарном дне, что и base.
export function parseHHmmToDate(s: string, base: Date): Date {
  const [h, m] = s.split(":").map(Number);
  const d = new Date(base);
  d.setHours(h, m, 0, 0);
  return d;
}

// R6: время начала блока. Фактическое начало; не раньше конца последнего блока.
export function computeStartTime(actualStart: Date, lastBlockEnd: Date | null): Date {
  if (lastBlockEnd && actualStart.getTime() < lastBlockEnd.getTime()) {
    return new Date(lastBlockEnd.getTime());
  }
  return new Date(actualStart.getTime());
}

// Собирает markdown-блок помодоро (формат ТЗ 4.2).
export function formatPomodoroBlock(start: Date, end: Date, lines: string[]): string {
  const head = `## 🍅 ${formatHHmm(start)} — ${formatHHmm(end)}\n\n`;
  const body = lines.map((l) => `- ${l.trim()}`).join("\n");
  return head + body + "\n";
}

// Конец (HH:mm) ПОСЛЕДНЕГО блока в заметке, либо null.
export function parseLastBlockEnd(content: string): string | null {
  const re = /^## 🍅 (\d{2}:\d{2}) — (\d{2}:\d{2})/gm;
  let m;
  let last: string | null = null;
  while ((m = re.exec(content)) !== null) {
    last = m[2];
  }
  return last;
}

// Все блоки помодоро в заметке.
export function parsePomodoroBlocks(content: string): PomodoroBlock[] {
  const re = /^## 🍅 (\d{2}:\d{2}) — (\d{2}:\d{2})/gm;
  const out: PomodoroBlock[] = [];
  let m;
  while ((m = re.exec(content)) !== null) {
    out.push({ start: m[1], end: m[2], heading: m[0] });
  }
  return out;
}

export interface DayStat {
  day: string; // "2026-11-10"
  count: number;
  minutes: number;
}

// Агрегирует помодоро по дням (отсортировано по дате).
export function aggregateStats(days: { day: string; content: string }[]): DayStat[] {
  return days
    .map((d) => {
      const blocks = parsePomodoroBlocks(d.content);
      let minutes = 0;
      for (const b of blocks) {
        const [sh, sm] = b.start.split(":").map(Number);
        const [eh, em] = b.end.split(":").map(Number);
        const diff = eh * 60 + em - (sh * 60 + sm);
        minutes += diff >= 0 ? diff : 0;
      }
      return { day: d.day, count: blocks.length, minutes };
    })
    .sort((a, b) => a.day.localeCompare(b.day));
}
```

```ts
import { pad2 } from "./core";

export type TimerMode = "work" | "break";
export type TimerState = "stopped" | "running" | "paused";

export interface TimerCallbacks {
  onWorkFinished: () => void;
  onTick: () => void;
}

export class PomodoroTimer {
  mode: TimerMode = "work";
  state: TimerState = "stopped";
  workMinutes = 30;
  breakMinutes = 5;
  private durationSec = 30 * 60;
  private remainingMs = 30 * 60 * 1000;
  private startedAt = 0;
  private interval: number | null = null;
  private cb: TimerCallbacks;

  constructor(cb: TimerCallbacks) {
    this.cb = cb;
  }

  // Изменить длительности можно только в состоянии «остановлен».
  setDurations(work: number, br: number) {
    if (this.state !== "stopped") return;
    this.workMinutes = Math.max(1, work);
    this.breakMinutes = Math.max(0, br);
    this.durationSec = (this.mode === "work" ? this.workMinutes : this.breakMinutes) * 60;
    this.remainingMs = this.durationSec * 1000;
    this.cb.onTick();
  }

  start() {
    if (this.state === "running") return;
    if (this.state === "stopped") {
      this.durationSec = (this.mode === "work" ? this.workMinutes : this.breakMinutes) * 60;
      this.remainingMs = this.durationSec * 1000;
      this.startedAt = Date.now();
    }
    this.state = "running";
    this.interval = window.setInterval(() => this.tick(), 500);
    this.cb.onTick();
  }

  pause() {
    if (this.state !== "running") return;
    this.state = "paused";
    if (this.interval !== null) {
      clearInterval(this.interval);
      this.interval = null;
    }
    this.cb.onTick();
  }

  stop() {
    if (this.interval !== null) {
      clearInterval(this.interval);
      this.interval = null;
    }
    this.state = "stopped";
    this.remainingMs = this.durationSec * 1000;
    this.cb.onTick();
  }

  // Момент запуска текущего рабочего интервала (настенные часы).
  get wallStart(): Date {
    return new Date(this.startedAt);
  }

  formatRemaining(): string {
    const total = Math.max(0, Math.round(this.remainingMs / 1000));
    const mm = Math.floor(total / 60);
    const ss = total % 60;
    return pad2(mm) + ":" + pad2(ss);
  }

  private tick() {
    if (this.state !== "running") return;
    const elapsed = Date.now() - this.startedAt;
    this.remainingMs = this.durationSec * 1000 - elapsed;
    if (this.remainingMs <= 0) {
      this.remainingMs = 0;
      this.finish();
    }
    this.cb.onTick();
  }

  private finish() {
    if (this.interval !== null) {
      clearInterval(this.interval);
      this.interval = null;
    }
    this.state = "stopped";
    this.remainingMs = this.durationSec * 1000;
    if (this.mode === "work") {
```

```ts
    const lines = this.textarea.value
      .split("\n")
      .map((s) => s.trim())

    this.resolveFn?.(lines);
    this.close();
  }

  }
```

```ts
import { TFile, moment } from "obsidian";
import type DnevnikPomodoro from "./main";
import { computeStartTime, formatPomodoroBlock, parseLastBlockEnd, parseHHmmToDate } from "./core";

// Дописывает блок помодоро в сегодняшнюю заметку; создаёт заметку по шаблону, если её нет.
export async function appendPomodoroBlock(plugin: DnevnikPomodoro, actualStart: Date, lines: string[]): Promise<void> {
  if (!lines.length) return;

  const settings = plugin.settings;
  const name = moment().format(settings.dailyNameFormat);
  const path = `${settings.dailyFolder}/${name}.md`;

  let file = plugin.app.vault.getAbstractFileByPath(path) as TFile | null;
  if (!file) {
    file = await createDailyNote(plugin, path);
  }

  const content = await plugin.app.vault.read(file);
  const lastEndStr = parseLastBlockEnd(content);
  const now = new Date();
  let lastEnd: Date | null = null;
  if (lastEndStr) lastEnd = parseHHmmToDate(lastEndStr, now);

  const start = computeStartTime(actualStart, lastEnd);
  const block = formatPomodoroBlock(start, now, lines);

  await plugin.app.vault.append(file, "\n" + block);
}

async function createDailyNote(plugin: DnevnikPomodoro, path: string): Promise<TFile> {
  const settings = plugin.settings;
  let template = "# \n";
  if (settings.dailyTemplate) {
    const tpl = plugin.app.vault.getAbstractFileByPath(settings.dailyTemplate) as TFile | null;
    if (tpl) {
      template = await plugin.app.vault.read(tpl);
      template = template.replace(/{{date:([^}]+)}}/g, (_, fmt: string) => moment().format(fmt));
    }
  }
  return plugin.app.vault.create(path, template);
}
```

```ts
import { Notice, Plugin } from "obsidian";
import { DnevnikSettings, DEFAULT_SETTINGS, DnevnikSettingTab } from "./settings";
import { PomodoroTimer } from "./timer";
import { CaptureModal } from "./capture-modal";
import { StatsModal } from "./stats-modal";
import { TIMER_VIEW_TYPE, TimerView } from "./timer-view";
import { appendPomodoroBlock } from "./daily";

export default class DnevnikPomodoro extends Plugin {
  settings: DnevnikSettings = DEFAULT_SETTINGS;
  timer!: PomodoroTimer;
  timerView: TimerView | null = null;
  private audioCtx: AudioContext | null = null;
  private statusBarEl: HTMLElement | null = null;

  async onload() {
    await this.loadSettings();
    this.timer = new PomodoroTimer({
      onWorkFinished: () => this.handleWorkFinished(),
      onTick: () => this.updateTimerUI(),
    });
    this.timer.setDurations(this.settings.workMinutes, this.settings.breakMinutes);

    this.registerView(TIMER_VIEW_TYPE, (leaf) => new TimerView(leaf, this));
    this.addRibbonIcon("timer", "Помодоро", () => this.toggleTimerView());
    this.addCommand({ id: "open-timer", name: "Открыть таймер", callback: () => this.toggleTimerView() });
    this.addCommand({ id: "capture", name: "Открыть сводку помодоро", callback: () => this.openCapture(new Date()) });
    this.addCommand({ id: "stats", name: "Открыть статистику", callback: () => new StatsModal(this.app, this).open() });
    this.addCommand({ id: "timer-start", name: "Таймер: старт", callback: () => this.timer.start() });
    this.addCommand({ id: "timer-pause", name: "Таймер: пауза", callback: () => this.timer.pause() });
    this.addCommand({ id: "timer-stop", name: "Таймер: стоп", callback: () => this.timer.stop() });

    this.addSettingTab(new DnevnikSettingTab(this.app, this));
    this.updateTimerUI();
  }

  onunload() {
    this.timer.stop();
  }

  async loadSettings() {
    this.settings = Object.assign({}, DEFAULT_SETTINGS, await this.loadData());
  }

  async saveSettings() {
    await this.saveData(this.settings);
  }

  private async toggleTimerView() {
    const existing = this.app.workspace.getLeavesOfType(TIMER_VIEW_TYPE);
    if (existing.length) {
      this.app.workspace.detachLeavesOfType(TIMER_VIEW_TYPE);
      this.timerView = null;
      return;
    }
    const leaf = this.app.workspace.getRightLeaf(false);
    await leaf.setViewState({ type: TIMER_VIEW_TYPE, active: true });
    this.app.workspace.revealLeaf(leaf);
    const leaves = this.app.workspace.getLeavesOfType(TIMER_VIEW_TYPE);
    this.timerView = leaves.length ? (leaves[0].view as TimerView) : null;
  }

  private async handleWorkFinished() {
    if (this.settings.sound) this.playBeep();
    if (this.settings.notification) {
      try {
        new Notification("Помодоро завершён", { body: "Время записать сводку" });
      } catch (e) {
        // уведомления недоступны — не критично
      }
    }
    await this.openCapture(this.timer.wallStart);
  }

  private playBeep() {
    try {
      if (!this.audioCtx) this.audioCtx = new AudioContext();
      const ctx = this.audioCtx;
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.frequency.value = 880;
      gain.gain.setValueAtTime(0.2, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.5);
      osc.start();
      osc.stop(ctx.currentTime + 0.5);
    } catch (e) {
      // звук недоступен — не критично
    }
  }

  private async openCapture(actualStart: Date) {
    const modal = new CaptureModal(this.app, this);
    const lines = await modal.openAndWait();
    if (lines && lines.length) {
      await appendPomodoroBlock(this, actualStart, lines);
      new Notice("Сводка записана");
      if (this.settings.autoBreak && this.timer.state === "stopped" && this.timer.mode === "break") {
        this.timer.start();
      }
    }
  }

  private updateTimerUI() {
    if (!this.statusBarEl) this.statusBarEl = this.addStatusBarItem();
    const label = this.timer.mode === "work" ? "🍅" : "☕";
    this.statusBarEl.setText(`${label} ${this.timer.formatRemaining()}`);
    if (this.timerView) this.timerView.render();
  }
}
```
